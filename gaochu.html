// 绑定事件 function bindEvents() { document.getElementById('addBtn').addEventListener('click', function() { openModal('add'); }); document.getElementById('refreshBtn').addEventListener('click', function() { loadCertData(); document.getElementById('searchInput').value = ''; }); document.getElementById('searchBtn').addEventListener('click', function() { const searchText = document.getElementById('searchInput').value.trim(); loadCertData(searchText); }); document.getElementById('searchInput').addEventListener('keypress', function(e) { if (e.key === 'Enter') { const searchText = this.value.trim(); loadCertData(searchText); } }); document.getElementById('exportExcel').addEventListener('click', exportToExcel); const closeButtons = document.querySelectorAll('.close, .closeModal'); closeButtons.forEach(btn => { btn.addEventListener('click', closeModal); }); document.getElementById('certForm').addEventListener('submit', async function(e) { e.preventDefault(); try { if (!GITHUB_CONFIG.token) { openTokenModal(async () => { await saveCert(); }); } else { await saveCert(); } } catch (error) { console.error('表单提交失败:', error); alert('表单提交失败: ' + error.message); } }); } // 绑定编辑和删除按钮事件 function bindEditDeleteEvents() { document.querySelectorAll('.edit-btn').forEach(btn => { btn.addEventListener('click', function() { const id = parseInt(this.getAttribute('data-id')); openModal('edit', id); }); }); document.querySelectorAll('.delete-btn').forEach(btn => { btn.addEventListener('click', async function() { const id = parseInt(this.getAttribute('data-id')); if (confirm('确定要删除这条记录吗？')) { try { if (!GITHUB_CONFIG.token) { openTokenModal(async () => { await deleteCert(id); }); } else { await deleteCert(id); } } catch (error) { console.error('删除失败:', error); alert('删除失败: ' + error.message); } } }); }); } // 打开模态框 async function openModal(mode, id = null) { const modal = document.getElementById('certModal'); const form = document.getElementById('certForm'); const title = document.getElementById('modalTitle'); if (mode === 'add') { title.textContent = '添加新证书'; form.reset(); document.getElementById('certId').value = ''; } else if (mode === 'edit' && id) { title.textContent = '编辑证书信息'; try { const data = await fetchFromGitHub(); const cert = data.find(item => item.id === id); if (cert) { Object.keys(cert).forEach(key => { const field = document.getElementById(key); if (field) { if (field.type === 'checkbox') { field.checked = cert[key]; } else { field.value = cert[key] || ''; } } }); document.getElementById('certId').value = id; } } catch (error) { console.error('加载编辑数据失败:', error); alert('加载编辑数据失败: ' + error.message); } } modal.style.display = 'block'; } // 关闭模态框 function closeModal() { document.getElementById('certModal').style.display = 'none'; } // 保存证书 async function saveCert() { try { const id = document.getElementById('certId').value; const isRed = document.getElementById('isRed').checked; const certData = { employeeId: document.getElementById('employeeId').value, employeeName: document.getElementById('employeeName').value, organization: document.getElementById('organization').value, workshop: document.getElementById('workshop').value, line: document.getElementById('line').value, certName: document.getElementById('certName').value, issuingAuthority: document.getElementById('issuingAuthority').value, certNumber: document.getElementById('certNumber').value, initialDate: document.getElementById('initialDate').value, validUntil: document.getElementById('validUntil').value, reviewDate: document.getElementById('reviewDate').value, workType: document.getElementById('workType').value, operationItem: document.getElementById('operationItem').value, actualReviewDate: document.getElementById('actualReviewDate').value, isRed: isRed }; let data = await fetchFromGitHub(); if (id) { const index = data.findIndex(item => item.id === parseInt(id)); if (index !== -1) { data[index] = { ...data[index], ...certData }; } } else { const newId = data.length > 0 ? Math.max(...data.map(item => item.id)) + 1 : 1; data.push({ id: newId, ...certData }); } await saveToGitHub(data); await loadCertData(); closeModal(); } catch (error) { console.error('保存证书失败:', error); alert('保存失败: ' + error.message); throw error; } } // 删除证书 async function deleteCert(id) { try { let data = await fetchFromGitHub(); data = data.filter(item => item.id !== id); await saveToGitHub(data); await loadCertData(); } catch (error) { console.error('删除证书失败:', error); alert('删除失败: ' + error.message); throw error; } } // 导出为Excel（中文列名） - 查看导出不需要令牌 async function exportToExcel() { try { const data = await fetchFromGitHub(); if (data.length === 0) { alert('没有可导出的数据'); return; } // 处理数据并映射中文列名 const processedData = data.map(item => { const newItem = {}; // 定义中英文字段映射 const fieldMap = { employeeId: '员工编号', employeeName: '员工姓名', organization: '组织', workshop: '中心站/车间', line: '所属线路', certName: '证书名称', issuingAuthority: '发证机关', certNumber: '证书编号', initialDate: '初次领证日期', validUntil: '有效期', reviewDate: '应复审日期', workType: '作业类别', operationItem: '操作项目', actualReviewDate: '实际复审日期' }; // 转换字段名 Object.keys(fieldMap).forEach(key => { newItem[fieldMap[key]] = item[key] || ''; }); return newItem; }); // 创建工作簿 const wb = XLSX.utils.book_new(); const ws = XLSX.utils.json_to_sheet(processedData); // 设置列宽 ws['!cols'] = [ {wch: 10}, {wch: 15}, {wch: 10}, {wch: 20}, {wch: 15}, {wch: 10}, {wch: 20}, {wch: 20}, {wch: 25}, {wch: 15}, {wch: 15}, {wch: 15}, {wch: 15}, {wch: 25} ]; XLSX.utils.book_append_sheet(wb, ws, "高处作业证数据"); // 导出文件 XLSX.writeFile(wb, `高处作业证_${new Date().toLocaleDateString()}.xlsx`); } catch (error) { console.error('导出Excel失败:', error); alert('导出失败: ' + error.message); } }
